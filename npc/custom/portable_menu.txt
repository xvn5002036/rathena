// Portable menu opened by the custom @menu atcommand.
-	script	PortableMenu	-1,{

// ============================================================
// Entry
// ============================================================
OnMenu:
	callsub S_ServerStatus;
	if (@PM_SettingsSync != 20260903) {
		@PM_ActualShowExp = getbattleflag("disp_experience");
		@PM_ActualNoAsk = 0;
		callsub S_ApplyPlayerSettings;
		@PM_SettingsSync = 20260903;
	}
	if (getgmlevel() > 0)
		goto OnGMMenu;
	goto OnPlayerMenu;

OnPCLoginEvent:
	@PM_ActualShowExp = getbattleflag("disp_experience");
	@PM_ActualNoAsk = 0;
	callsub S_ApplyPlayerSettings;
	@PM_SettingsSync = 20260903;
	end;

// ============================================================
// Player menu
// ============================================================
OnPlayerMenu:
	.@back$ = (getgmlevel() > 0) ? "^4B0082◆^000000 返回管理員主選單" : "^555555◆ 關閉選單^000000";
	.@menu$ = "^8B0000◆^000000 冒險支援:^000080◆^000000 補給商店:";
	.@type[0] = 1;
	.@type[1] = 2;
	.@count = 2;
	if ($PMServiceActive) {
		.@menu$ += "^8B4513◆^000000 便捷服務:";
		.@type[.@count++] = 3;
	}
	if ($PMRoleActive) {
		.@menu$ += "^000080◆^000000 角色服務:";
		.@type[.@count++] = 4;
	}
	if ($PWRefine || $PWGrade || $PWMerge || $PWEnchant || $PWSocket || $PWCardRemove || $PWCardMove || $PWGradeExchange || $PWRepair) {
		.@menu$ += "^4B0082◆^000000 裝備工坊:";
		.@type[.@count++] = 5;
	}
	if ($PMTravelActive) {
		.@menu$ += "^006400◆^000000 快速傳送:";
		.@type[.@count++] = 6;
	}
	.@menu$ += "^8B4513◆^000000 個人設定:" + .@back$;
	.@type[.@count++] = 7;
	.@choice = select(.@menu$);
	if (.@choice > .@count) {
		if (getgmlevel() > 0)
			goto OnGMMenu;
		close;
	}
	switch(.@type[.@choice - 1]) {
	case 1: goto OnPlayerSupport;
	case 2: goto OnPlayerStore;
	case 3: goto OnPlayerConvenience;
	case 4: goto OnPlayerRoleService;
	case 5: goto OnPlayerWorkshop;
	case 6: goto OnPlayerTravel;
	case 7: goto OnPlayerSettings;
	}
	goto OnPlayerMenu;

OnPlayerSupport:
	switch(select("^8B0000◆^000000 生命恢復【" + $PMHealFee + "】:^000080◆^000000 儲點返回【" + $PMReturnFee + "】:^8B4513◆^000000 隨身倉庫【" + $PMStorageFee + "】:^006400◆^000000 儲點登錄【" + $PMSaveFee + "】:^4B0082◆^000000 公會倉庫【" + $PMGuildStorageFee + "】:^555555◇ 返回選單^000000")) {
	case 1:
		if (Hp >= MaxHp && Sp >= MaxSp) { dispbottom "目前生命與魔力都是全滿狀態。"; goto OnPlayerSupport; }
		if (select("^8B0000◆^000000 支付 " + $PMHealFee + " 遊戲幣並完全恢復:^555555◇ 取消^000000") != 1) goto OnPlayerSupport;
		if (Zeny < $PMHealFee) { dispbottom "遊戲幣不足。"; goto OnPlayerSupport; }
		Zeny -= $PMHealFee;
		percentheal 100, 100;
		goto OnPlayerSupport;
	case 2:
		if (select("^000080◆^000000 支付 " + $PMReturnFee + " 遊戲幣並返回儲點:^555555◇ 取消^000000") != 1) goto OnPlayerSupport;
		if (Zeny < $PMReturnFee) { dispbottom "遊戲幣不足。"; goto OnPlayerSupport; }
		Zeny -= $PMReturnFee;
		warp "SavePoint", 0, 0;
		end;
	case 3:
		if (Zeny < $PMStorageFee) { dispbottom "遊戲幣不足。"; goto OnPlayerSupport; }
		Zeny -= $PMStorageFee;
		openstorage;
		end;
	case 4:
		getmapxy(.@map$,.@x,.@y,BL_PC);
		if (select("^006400◆^000000 支付 " + $PMSaveFee + " 遊戲幣並登錄目前位置:^555555◇ 取消^000000") != 1) goto OnPlayerSupport;
		if (Zeny < $PMSaveFee) { dispbottom "遊戲幣不足。"; goto OnPlayerSupport; }
		Zeny -= $PMSaveFee;
		savepoint .@map$,.@x,.@y;
		dispbottom "目前位置已登錄為儲存地點。";
		goto OnPlayerSupport;
	case 5:
		if (getcharid(2) < 1) { dispbottom "目前角色尚未加入公會。"; goto OnPlayerSupport; }
		if (Zeny < $PMGuildStorageFee) { dispbottom "遊戲幣不足。"; goto OnPlayerSupport; }
		Zeny -= $PMGuildStorageFee;
		.@result = guildopenstorage();
		if (.@result != GSTORAGE_OPEN) {
			Zeny += $PMGuildStorageFee;
			dispbottom "無法開啟公會倉庫，費用已退還。";
			goto OnPlayerSupport;
		}
		end;
	default:
		goto OnPlayerMenu;
	}

OnPlayerStore:
	.@menu$ = "^006400◆^000000 藥水商店 ^8B0000【加價五成】^000000:";
	.@type[0] = 1;
	.@count = 1;
	if ($PMSupplyActive) {
		.@menu$ += "^8B0000◆^000000 活動補給 ^8B4513【限領一次】^000000:";
		.@type[.@count++] = 2;
	}
	.@menu$ += "^555555◇ 返回選單^000000";
	.@choice = select(.@menu$);
	if (.@choice > .@count)
		goto OnPlayerMenu;
	if (.@type[.@choice - 1] == 1)
		goto OnPlayerPotionShop;
	goto OnPlayerSupply;

OnPlayerConvenience:
	if (!$PMServiceActive)
		goto OnPlayerMenu;
	switch(select("^8B4513◆^000000 全包鑑定【每件 " + $PMIdentifyFee + " 遊戲幣】:^4B0082◆^000000 狀態重整【費用 " + $PMPurifyFee + " 遊戲幣】:^000080◆^000000 角色資訊:^555555◇ 返回選單^000000")) {
	case 1:
		.@count = identifyall(0);
		if (.@count < 1) {
			dispbottom "背包中沒有需要鑑定的物品。";
			goto OnPlayerConvenience;
		}
		.@fee = .@count * $PMIdentifyFee;
		if (select("^8B4513◆^000000 鑑定 " + .@count + " 件【合計 " + .@fee + " 遊戲幣】:^555555◇ 取消^000000") != 1)
			goto OnPlayerConvenience;
		.@count = identifyall(0);
		.@fee = .@count * $PMIdentifyFee;
		if (.@count < 1 || Zeny < .@fee) {
			dispbottom "物品狀態已改變，或遊戲幣不足。";
			goto OnPlayerConvenience;
		}
		Zeny -= .@fee;
		identifyall(1);
		dispbottom "全包鑑定完成，共處理 " + .@count + " 件物品。";
		goto OnPlayerConvenience;
	case 2:
		if (select("^4B0082◆^000000 清除所有增益與異常狀態【" + $PMPurifyFee + " 遊戲幣】:^555555◇ 取消^000000") != 1)
			goto OnPlayerConvenience;
		if (Zeny < $PMPurifyFee) {
			dispbottom "遊戲幣不足，需要 " + $PMPurifyFee + " 遊戲幣。";
			goto OnPlayerConvenience;
		}
		Zeny -= $PMPurifyFee;
		sc_end SC_ALL;
		dispbottom "狀態重整完成，所有增益與異常狀態已清除。";
		goto OnPlayerConvenience;
	case 3:
		goto OnPlayerInformation;
	default:
		goto OnPlayerMenu;
	}

OnPlayerInformation:
	.@weight = (MaxWeight > 0) ? Weight * 100 / MaxWeight : 0;
	select("^8B4513◆^000000 角色名稱【" + strcharinfo(0) + "】:^000080◆^000000 基本等級【" + BaseLevel + "】:^4B0082◆^000000 職業等級【" + JobLevel + "】:^006400◆^000000 持有金額【" + Zeny + "】:^8B0000◆^000000 負重比例【" + .@weight + "%】:^555555◇ 返回便捷服務^000000");
	goto OnPlayerConvenience;

OnPlayerRoleService:
	if (!$PMRoleActive)
		goto OnPlayerMenu;
	switch(select("^8B0000◆^000000 素質重置【" + $PMStatResetFee + "】:^4B0082◆^000000 技能重置【" + $PMSkillResetFee + "】:^8B4513◆^000000 髮型變更【" + $PMHairFee + "】:^006400◆^000000 髮色變更【" + $PMHairColorFee + "】:^000080◆^000000 服裝染色【" + $PMClothFee + "】:^555555◇ 返回選單^000000")) {
	case 1:
		if (select("^8B0000◆^000000 確定重置全部素質點數【不可復原】:^555555◇ 取消^000000") != 1) goto OnPlayerRoleService;
		if (Zeny < $PMStatResetFee) { dispbottom "遊戲幣不足。"; goto OnPlayerRoleService; }
		Zeny -= $PMStatResetFee;
		resetstatus;
		dispbottom "素質重置完成。";
		goto OnPlayerRoleService;
	case 2:
		if (select("^4B0082◆^000000 確定重置全部技能點數【不可復原】:^555555◇ 取消^000000") != 1) goto OnPlayerRoleService;
		if (Zeny < $PMSkillResetFee) { dispbottom "遊戲幣不足。"; goto OnPlayerRoleService; }
		Zeny -= $PMSkillResetFee;
		resetskill;
		dispbottom "技能重置完成。";
		goto OnPlayerRoleService;
	case 3:
		callsub S_PlayerStylePreview,LOOK_HAIR,getbattleflag("max_hair_style"),$PMHairFee,"髮型";
		goto OnPlayerRoleService;
	case 4:
		callsub S_PlayerStylePreview,LOOK_HAIR_COLOR,getbattleflag("max_hair_color"),$PMHairColorFee,"髮色";
		goto OnPlayerRoleService;
	case 5:
		callsub S_PlayerStylePreview,LOOK_CLOTHES_COLOR,getbattleflag("max_cloth_color"),$PMClothFee,"服裝色彩";
		goto OnPlayerRoleService;
	default:
		goto OnPlayerMenu;
	}

S_PlayerStylePreview:
	.@look = getarg(0);
	.@max = getarg(1);
	.@fee = getarg(2);
	.@title$ = getarg(3);
	.@original = getlook(.@look);
	.@style = (.@original >= 1 && .@original <= .@max) ? .@original : 1;
	if (.@max < 1) {
		dispbottom "目前客戶端沒有可用的" + .@title$ + "選項。";
		return;
	}
	setlook .@look,.@style;
	while (1) {
		.@choice = select("^006400◆^000000 下一款式【目前 " + .@style + "／" + .@max + "】:^8B4513◆^000000 上一款式【目前 " + .@style + "／" + .@max + "】:^000080◆^000000 指定編號:^4B0082◆^000000 確認套用【費用 " + .@fee + "】:^555555◇ 取消變更^000000");
		switch(.@choice) {
		case 1:
			.@style = (.@style < .@max) ? .@style + 1 : 1;
			setlook .@look,.@style;
			break;
		case 2:
			.@style = (.@style > 1) ? .@style - 1 : .@max;
			setlook .@look,.@style;
			break;
		case 3:
			input .@style,1,.@max;
			setlook .@look,.@style;
			break;
		case 4:
			if (.@style == .@original) {
				dispbottom .@title$ + "沒有變更，因此不收取費用。";
				return;
			}
			if (Zeny < .@fee) {
				setlook .@look,.@original;
				dispbottom "遊戲幣不足，已恢復原本的" + .@title$ + "。";
				return;
			}
			Zeny -= .@fee;
			dispbottom .@title$ + "變更完成，已支付 " + .@fee + " 遊戲幣。";
			return;
		default:
			setlook .@look,.@original;
			dispbottom "已取消變更並恢復原本的" + .@title$ + "。";
			return;
		}
	}

OnPlayerPotionShop:
	callshop "PortablePotionShop", 1;
	end;

OnPlayerSupply:
	if (!$PMSupplyActive || $PMSupplyRound < 1)
		goto OnPlayerMenu;
	if ($PMSupplyName$ != "")
		dispbottom "目前活動：" + $PMSupplyName$;
	.@claimed$ = (#PMSupplyClaimRound == $PMSupplyRound) ? "^8B0000已領取^000000" : "^006400尚未領取^000000";
	switch(select("^8B0000★^000000 領取活動補給【費用 " + $PMSupplyFee + " 遊戲幣】:^000080◆^000000 查看補給內容:領取狀態【" + .@claimed$ + "】:^555555◇ 返回玩家主選單^000000")) {
	case 1:
		if (#PMSupplyClaimRound == $PMSupplyRound) {
			dispbottom "這一期活動補給已經領取過了。";
			goto OnPlayerSupply;
		}
		if ($PMSupplyCount < 1 || $PMSupplyFee < 1) {
			dispbottom "目前活動補給尚未設定完成。";
			goto OnPlayerSupply;
		}
		if (Zeny < $PMSupplyFee) {
			dispbottom "遊戲幣不足，需要 " + $PMSupplyFee + " 遊戲幣。";
			goto OnPlayerSupply;
		}
		.@total_weight = 0;
		for (.@i = 0; .@i < $PMSupplyCount; .@i++) {
			if (!checkweight($PMSupplyItem[.@i],$PMSupplyAmount[.@i])) {
				dispbottom "負重或物品欄空間不足，無法領取。";
				goto OnPlayerSupply;
			}
			.@total_weight += getiteminfo($PMSupplyItem[.@i],ITEMINFO_WEIGHT) * $PMSupplyAmount[.@i];
		}
		if (Weight + .@total_weight > MaxWeight) {
			dispbottom "總負重不足，請整理背包後再領取。";
			goto OnPlayerSupply;
		}
		if (select("^8B0000確定支付 " + $PMSupplyFee + " 遊戲幣並領取^000000:^555555取消^000000") != 1)
			goto OnPlayerSupply;
		if (!$PMSupplyActive || #PMSupplyClaimRound == $PMSupplyRound || Zeny < $PMSupplyFee)
			goto OnPlayerSupply;
		Zeny -= $PMSupplyFee;
		for (.@i = 0; .@i < $PMSupplyCount; .@i++)
			getitem $PMSupplyItem[.@i],$PMSupplyAmount[.@i];
		#PMSupplyClaimRound = $PMSupplyRound;
		dispbottom "活動補給領取成功，已支付 " + $PMSupplyFee + " 遊戲幣。";
		goto OnPlayerSupply;
	case 2:
		dispbottom "【" + $PMSupplyName$ + "】活動補給內容：";
		for (.@i = 0; .@i < $PMSupplyCount; .@i++) {
			.@name$ = "";
			query_sql "SELECT `display_name` FROM `client_item_names` WHERE `item_id` = " + $PMSupplyItem[.@i], .@name$;
			if (.@name$ == "") .@name$ = getitemname($PMSupplyItem[.@i]);
			dispbottom "◆ " + .@name$ + " × " + $PMSupplyAmount[.@i];
		}
		goto OnPlayerSupply;
	case 3:
		if (#PMSupplyClaimRound == $PMSupplyRound)
			dispbottom "本期已領取。";
		else
			dispbottom "本期尚未領取。";
		goto OnPlayerSupply;
	default:
		goto OnPlayerMenu;
	}

OnPlayerTravel:
	switch(select("^8B4513◆^000000 城鎮傳送【" + $PMTravelTownFee + " 遊戲幣】:^006400◆^000000 練功區傳送【" + $PMTravelFieldFee + " 遊戲幣】:^4B0082◆^000000 地下城傳送【" + $PMTravelDungeonFee + " 遊戲幣】:^555555◇ 返回玩家主選單^000000")) {
	case 1: goto OnPlayerTravelTown;
	case 2: goto OnPlayerTravelField;
	case 3: goto OnPlayerTravelDungeon;
	default: goto OnPlayerMenu;
	}

OnPlayerTravelTown:
	.@c = select("^8B4513◆^000000 普隆德拉:^000080◆^000000 吉芬:^006400◆^000000 斐揚:^8B4513◆^000000 夢羅克:^000080◆^000000 艾爾貝塔:^4B0082◆^000000 艾爾帕蘭:^8B4513◆^000000 朱諾:^8B0000◆^000000 里希塔樂鎮:^006400◆^000000 艾音布羅克:^4B0082◆^000000 拉赫:^555555◇ 返回傳送分類^000000");
	if (.@c == 11) goto OnPlayerTravel;
	setarray .@map$[1],"prontera","geffen","payon","morocc","alberta","aldebaran","yuno","lighthalzen","einbroch","rachel";
	setarray .@x[1],156,119,162,160,117,140,157,158,64,130;
	setarray .@y[1],191,59,233,94,56,131,51,92,200,110;
	callsub S_PlayerPaidWarp,.@map$[.@c],.@x[.@c],.@y[.@c],$PMTravelTownFee;
	goto OnPlayerTravelTown;

OnPlayerTravelField:
	.@c = select("^006400◆^000000 波利草原:^006400◆^000000 斐揚樹林:^8B4513◆^000000 夢羅克沙漠:^8B0000◆^000000 獸人地下洞窟:^4B0082◆^000000 斐揚地下洞穴:^000080◆^000000 普隆德拉地下水道:^8B4513◆^000000 妙勒尼山脈:^555555◇ 返回傳送分類^000000");
	if (.@c == 8) goto OnPlayerTravel;
	setarray .@map$[1],"prt_fild08","pay_fild08","moc_fild12","orcsdun01","pay_dun00","prt_sewb1","mjolnir_02";
	setarray .@x[1],170,80,170,32,21,131,90;
	setarray .@y[1],371,350,280,170,183,248,190;
	callsub S_PlayerPaidWarp,.@map$[.@c],.@x[.@c],.@y[.@c],$PMTravelFieldFee;
	goto OnPlayerTravelField;

OnPlayerTravelDungeon:
	.@c = select("^4B0082◆^000000 吉芬地下城:^8B4513◆^000000 金字塔迷宮:^8B4513◆^000000 史芬克斯:^000080◆^000000 海底洞窟:^555555◆^000000 古城入口:^8B4513◆^000000 鐘樓一樓:^4B0082◆^000000 死亡之塔:^000080◆^000000 冰洞穴:^8B0000◆^000000 火山洞窟:^006400◆^000000 生體研究所:^555555◇ 返回傳送分類^000000");
	if (.@c == 11) goto OnPlayerTravel;
	setarray .@map$[1],"gef_dun00","moc_pryd01","in_sphinx1","iz_dun00","glast_01","c_tower1","tha_t01","ice_dun01","thor_v01","lhz_dun01";
	setarray .@x[1],104,192,288,168,199,199,150,157,21,149;
	setarray .@y[1],99,9,9,168,294,159,74,14,227,291;
	callsub S_PlayerPaidWarp,.@map$[.@c],.@x[.@c],.@y[.@c],$PMTravelDungeonFee;
	goto OnPlayerTravelDungeon;

S_PlayerPaidWarp:
	.@map$ = getarg(0); .@x = getarg(1); .@y = getarg(2); .@fee = getarg(3);
	if (!$PMTravelActive || .@fee < 1) return;
	if (Zeny < .@fee) { dispbottom "遊戲幣不足，需要 " + .@fee + " 遊戲幣。"; return; }
	if (select("^8B0000◆^000000 確定支付 " + .@fee + " 遊戲幣並傳送:^555555◇ 取消^000000") != 1) return;
	if (!$PMTravelActive || Zeny < .@fee) return;
	Zeny -= .@fee;
	warp .@map$,.@x,.@y;
	end;

OnPlayerWorkshop:
	.@menu$ = "";
	.@count = 0;
	if ($PWRefine) { .@menu$ += "^8B4513◆^000000 裝備精煉:"; .@type[.@count++] = 1; }
	if ($PWGrade) { .@menu$ += "^8B0000◆^000000 裝備階級強化:"; .@type[.@count++] = 2; }
	if ($PWMerge) { .@menu$ += "^000080◆^000000 物品合成:"; .@type[.@count++] = 3; }
	if ($PWEnchant) { .@menu$ += "^4B0082★^000000 台版風格裝備附魔:"; .@type[.@count++] = 9; }
	if ($PWSocket) { .@menu$ += "^000080◆^000000 裝備打洞:"; .@type[.@count++] = 4; }
	if ($PWCardRemove) { .@menu$ += "^8B0000◆^000000 卡片拆卸:"; .@type[.@count++] = 5; }
	if ($PWCardMove) { .@menu$ += "^006400◆^000000 購買移卡券:"; .@type[.@count++] = 6; }
	if ($PWGradeExchange) { .@menu$ += "^8B4513◆^000000 階級材料交換:"; .@type[.@count++] = 7; }
	if ($PWRepair) { .@menu$ += "^006400◆^000000 修理損壞裝備:"; .@type[.@count++] = 8; }
	if (!.@count)
		goto OnPlayerMenu;
	.@menu$ += "^555555◇ 返回玩家主選單^000000";
	.@choice = select(.@menu$);
	if (.@choice > .@count)
		goto OnPlayerMenu;
	switch(.@type[.@choice - 1]) {
	case 1:
		refineui();
		end;
	case 2:
		enchantgradeui();
		end;
	case 3:
		mergeitem;
		goto OnPlayerWorkshop;
	case 4:
		goto OnPlayerSocket;
	case 5:
		goto OnPlayerCardRemove;
	case 6:
		callshop "change_slot#slo";
		end;
	case 7:
		callshop "EnchantGradeExchange";
		end;
	case 8:
		goto OnPlayerRepair;
	case 9:
		goto OnPlayerEnchant;
	default:
		goto OnPlayerMenu;

S_SocketApply:
	.@src = getarg(0); .@dst = getarg(1); .@low = getarg(2); .@high = getarg(3);
	.@fee = getarg(4) * 1000; .@mat1 = getarg(5); .@amt1 = getarg(6);

	.@confirm$ = getitemname(.@src) + " → " + getitemname(.@dst) + "｜費用 " + .@fee + " Zeny｜確定:";
	.@confirm$ += "取消";
	if (select(.@confirm$) != 1) return;
	if (!checkweight(.@dst,1)) { dispbottom "負重或物品欄空間不足。"; return; }
	if (Zeny < .@fee || countitem(.@src) < 1 || countitem(.@mat1) < .@amt1 || (.@mat2 && countitem(.@mat2) < .@amt2)) {
		dispbottom "Zeny、原裝備或打洞材料不足。";
		return;
	}
	delitem .@src,1;
	delitem .@mat1,.@amt1;
	if (.@mat2) delitem .@mat2,.@amt2;
	Zeny -= .@fee;
	.@roll = rand(1,100);
	if (.@roll > .@low && .@roll < .@high) {
		getitem .@dst,1;
		dispbottom "打洞成功：已取得 " + getitemname(.@dst) + "。";
	}
	else
		dispbottom "打洞失敗：裝備、材料與費用已消耗。";
	return;

	.@count = 0;
	while (getbrokenid(.@count + 1)) .@count++;

	.@fee = .@count * 5000;

	.@verify = 0;
	while (getbrokenid(.@verify + 1)) .@verify++;
	if (.@verify != .@count) { dispbottom "損壞裝備數量已變更，請重新操作。"; goto OnPlayerWorkshop; }

	Zeny -= .@fee;
	while (.@count) { repair(.@count); .@count--; }
	dispbottom "裝備修理完成，已支付 " + .@fee + " Zeny。";
	goto OnPlayerWorkshop;

OnPlayerPassword:
	dispbottom "請輸入目前登入密碼；輸入內容不會顯示在聊天紀錄。";
	input .@old$;
	dispbottom "請輸入 1 至 23 位英文字母或數字的新密碼。";
	input .@new$;
	dispbottom "請再次輸入新密碼。";
	input .@confirm$;
	if (getstrlen(.@new$) < 1 || getstrlen(.@new$) > 23 || !preg_match("^[A-Za-z0-9]+$",.@new$)) {
		dispbottom "新密碼格式不符，限 1 至 23 位英文字母或數字。";
		goto OnPlayerPasswordDone;
	}
	if (strcmp(.@new$,.@confirm$) != 0) {
		dispbottom "兩次輸入的新密碼不同，密碼沒有變更。";
		goto OnPlayerPasswordDone;
	}
	if (strcmp(.@old$,.@new$) == 0) {
		dispbottom "新密碼不能和目前密碼相同。";
		goto OnPlayerPasswordDone;
	}
	.@account = getcharid(3);
	.@stored$ = "";
	query_sql "SELECT `user_pass` FROM `login` WHERE `account_id` = " + .@account + " LIMIT 1",.@stored$;
	.@old_md5$ = md5(.@old$);
	if (strcmp(.@stored$,.@old$) != 0 && strcmp(.@stored$,.@old_md5$) != 0) {
		dispbottom "目前密碼不正確，密碼沒有變更。";
		goto OnPlayerPasswordDone;
	}
	.@save$ = (strcmp(.@stored$,.@old_md5$) == 0) ? md5(.@new$) : .@new$;
	.@escaped$ = escape_sql(.@save$);
	query_sql "UPDATE `login` SET `user_pass` = '" + .@escaped$ + "' WHERE `account_id` = " + .@account + " LIMIT 1";
	.@verify$ = "";
	query_sql "SELECT `user_pass` FROM `login` WHERE `account_id` = " + .@account + " LIMIT 1",.@verify$;
	if (strcmp(.@verify$,.@save$) == 0)
		dispbottom "登入密碼變更成功，下次登入請使用新密碼。";
	else
		dispbottom "密碼更新失敗，原密碼仍然有效。";

OnPlayerPasswordDone:
	.@old$ = "";
	.@new$ = "";
	.@confirm$ = "";
	.@stored$ = "";
	.@old_md5$ = "";
	.@save$ = "";
	.@escaped$ = "";
	.@verify$ = "";
	goto OnPlayerSettings;
