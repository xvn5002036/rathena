// Portable menu opened by the custom @menu atcommand.
-	script	PortableMenu	-1,{

// ============================================================
// Entry
// ============================================================
OnMenu:
	if (getgmlevel() > 0)
		goto OnGMMenu;
	goto OnPlayerMenu;

// ============================================================
// Player menu
// ============================================================
OnPlayerMenu:
	switch(select("完全恢復 HP / SP:返回儲存點:個人倉庫:便捷設定:關閉菜單")) {
	case 1:
		percentheal 100, 100;
		goto OnPlayerMenu;
	case 2:
		warp "SavePoint", 0, 0;
		end;
	case 3:
		openstorage;
		end;
	case 4:
		goto OnPlayerSettings;
	default:
		close;
	}

OnPlayerSettings:
	.@menu$ = "自動拾取【" + (#PM_Autoloot ? "^00AA00開啟^000000" : "^FF5555關閉^000000") + "】:";
	.@menu$ += "經驗值顯示【" + (#PM_ShowExp ? "^00AA00開啟^000000" : "^FF5555關閉^000000") + "】:";
	.@menu$ += "Zeny 顯示【" + (#PM_ShowZeny ? "^00AA00開啟^000000" : "^FF5555關閉^000000") + "】:";
	.@menu$ += "拒絕邀請【" + (#PM_NoAsk ? "^00AA00開啟^000000" : "^FF5555關閉^000000") + "】:";
	.@menu$ += "返回玩家主菜單";

	switch(select(.@menu$)) {
	case 1:
		#PM_Autoloot = !#PM_Autoloot;
		.@command$ = #PM_Autoloot ? "@autoloot 100" : "@autoloot 0";
		atcommand .@command$;
		goto OnPlayerSettings;
	case 2:
		#PM_ShowExp = !#PM_ShowExp;
		atcommand "@showexp";
		goto OnPlayerSettings;
	case 3:
		#PM_ShowZeny = !#PM_ShowZeny;
		atcommand "@showzeny";
		goto OnPlayerSettings;
	case 4:
		#PM_NoAsk = !#PM_NoAsk;
		atcommand "@noask";
		goto OnPlayerSettings;
	default:
		goto OnPlayerMenu;
	}

// ============================================================
// GM main menu
// ============================================================
OnGMMenu:
	.@menu$ = "自身功能:快速傳送:玩家工具:倉庫功能:";
	.@count = 4;
	if (getgmlevel() >= 99) {
		.@menu$ += "角色管理:管理員工具:";
		.@count = 6;
	}
	.@menu$ += "關閉菜單";
	.@choice = select(.@menu$);

	if (.@choice > .@count)
		close;

	switch(.@choice) {
	case 1: goto OnGMSelfTools;
	case 2: goto OnGMWarpMenu;
	case 3: goto OnGMPlayerTools;
	case 4: goto OnGMStorageMenu;
	case 5: goto OnGMCharacterSetup;
	case 6: goto OnGMAdminTools;
	}
	goto OnGMMenu;

// ============================================================
// GM character management
// ============================================================
OnGMCharacterSetup:
	if (getgmlevel() < 99)
		close;

	switch(select("金錢／點數設定:職業切換:坐騎／夥伴:學會全部技能:套用 GM 素質:領取 GM 特殊裝備:返回 GM 主菜單")) {
	case 1:
		goto OnGMCurrency;
	case 2:
		goto OnGMJobMenu;
	case 3:
		goto OnGMCompanionMenu;
	case 4:
		atcommand "@allskill";
		goto OnGMCharacterSetup;
	case 5:
		statusup2 bStr, 130 - readparam(bStr);
		statusup2 bAgi, 130 - readparam(bAgi);
		statusup2 bVit, 130 - readparam(bVit);
		statusup2 bInt, 130 - readparam(bInt);
		statusup2 bDex, 130 - readparam(bDex);
		statusup2 bLuk, 130 - readparam(bLuk);
		statusup2 bPow, 100 - readparam(bPow);
		statusup2 bSta, 100 - readparam(bSta);
		statusup2 bWis, 100 - readparam(bWis);
		statusup2 bSpl, 100 - readparam(bSpl);
		statusup2 bCon, 100 - readparam(bCon);
		statusup2 bCrt, 100 - readparam(bCrt);
		goto OnGMCharacterSetup;
	case 6:
		getitem 50000, 1;
		getitem 50001, 1;
		getitem 50002, 1;
		getitem 50003, 1;
		getitem 50004, 1;
		getitem 50005, 1;
		getitem 50006, 1;
		getitem 50007, 1;
		getitem 50008, 1;
		getitem 50009, 1;
		getitem 50010, 1;
		getitem 50011, 1;
		getitem 50012, 1;
		goto OnGMCharacterSetup;
	default:
		goto OnGMMenu;
	}

OnGMCurrency:
	switch(select("Zeny【" + Zeny + "】:商城點數【" + #CASHPOINTS + "】:Kafra 點數【" + #KAFRAPOINTS + "】:返回角色管理")) {
	case 1:
		input .@value, 0, 2147483647;
		Zeny = .@value;
		goto OnGMCurrency;
	case 2:
		input .@value, 0, 2147483647;
		#CASHPOINTS = .@value;
		goto OnGMCurrency;
	case 3:
		input .@value, 0, 2147483647;
		#KAFRAPOINTS = .@value;
		goto OnGMCurrency;
	default:
		goto OnGMCharacterSetup;
	}

// ============================================================
// GM companions and mounts
// ============================================================
OnGMCompanionMenu:
	.@menu$ = "";
	.@count = 0;
	deletearray .@type;

	if (getskilllv("MC_PUSHCART") > 0) {
		.@menu$ += "推車【" + (checkcart() ? "^00AA00使用中^000000" : "^FF5555未使用^000000") + "】:";
		.@type[.@count] = 1;
		.@count++;
	}

	if (
		getskilllv("HT_FALCON") > 0 &&
		Class != JOB_RANGER &&
		Class != JOB_WINDHAWK
	) {
		.@menu$ += "獵鷹【" + (checkfalcon() ? "^00AA00使用中^000000" : "^FF5555未使用^000000") + "】:";
		.@type[.@count] = 2;
		.@count++;
	}

	if (
		getskilllv("KN_RIDING") > 0 &&
		Class != JOB_RUNE_KNIGHT &&
		Class != JOB_DRAGON_KNIGHT
	) {
		.@menu$ += "騎乘獸【" + (checkriding() ? "^00AA00騎乘中^000000" : "^FF5555未騎乘^000000") + "】:";
		.@type[.@count] = 3;
		.@count++;
	}

	if (
		Class == JOB_RUNE_KNIGHT ||
		Class == JOB_DRAGON_KNIGHT
	) {
		.@menu$ += "龍【" + (checkdragon() ? "^00AA00騎乘中^000000" : "^FF5555未騎乘^000000") + "】:";
		.@type[.@count] = 4;
		.@count++;
	}

	if (
		Class == JOB_MECHANIC ||
		Class == JOB_MEISTER
	) {
		.@menu$ += "魔導機甲【" + (checkmadogear() ? "^00AA00搭乘中^000000" : "^FF5555未搭乘^000000") + "】:";
		.@type[.@count] = 5;
		.@count++;
	}

	if (
		Class == JOB_RANGER ||
		Class == JOB_WINDHAWK
	) {
		.@menu$ += "遊俠夥伴【" + (checkwug() ? "^00AA00狼^000000" : (checkfalcon() ? "^00AA00獵鷹^000000" : "^FF5555無^000000")) + "】:";
		.@type[.@count] = 6;
		.@count++;
	}

	.@menu$ += "返回 GM 角色設定";
	.@sel = select(.@menu$);

	if (.@sel > .@count) {
		goto OnGMCharacterSetup;
	}

	switch(.@type[.@sel - 1]) {
	case 1: goto OnGMCartToggle;
	case 2: goto OnGMFalconToggle;
	case 3: goto OnGMRidingToggle;
	case 4: goto OnGMDragonToggle;
	case 5: goto OnGMMadoToggle;
	case 6: goto OnGMRangerPet;
	}

	goto OnGMCompanionMenu;

OnGMCartToggle:
	if (checkcart())
		setcart 0;
	else
		setcart;
	goto OnGMCompanionMenu;

OnGMFalconToggle:
	if (checkfalcon()) {
		setfalcon 0;
	}
	else if (!checkwug()) {
		setfalcon;
	}
	goto OnGMCompanionMenu;

OnGMRidingToggle:
	if (checkriding())
		setriding 0;
	else
		setriding;
	goto OnGMCompanionMenu;

OnGMDragonToggle:
	if (checkdragon()) {
		setdragon 0;
		goto OnGMCompanionMenu;
	}

	switch(select("綠龍:棕龍:灰龍:藍龍:紅龍:返回")) {
	case 1: setdragon 1; break;
	case 2: setdragon 2; break;
	case 3: setdragon 3; break;
	case 4: setdragon 4; break;
	case 5: setdragon 5; break;
	}
	goto OnGMCompanionMenu;

OnGMMadoToggle:
	if (checkmadogear())
		setmadogear 0;
	else
		setmadogear;
	goto OnGMCompanionMenu;

OnGMRangerPet:
	.@falcon$ = "獵鷹【" + (checkfalcon() ? "^00AA00使用中^000000" : "^FF5555未使用^000000") + "】";
	.@wug$ = "狼笛【" + (countitem(6124) ? "^00AA00已持有^000000" : "^FF5555未持有^000000") + "】";

			setfalcon 0;
		else if (!checkwug())
			setfalcon;
		goto OnGMRangerPet;

	case 2:
		if (!checkfalcon() && !checkwug() && !countitem(6124))

		goto OnGMRangerPet;

	default:
		goto OnGMCompanionMenu;

	case 4: goto OnGMJobThird;

OnGMJobCommand:
	input .@job_id, 0, 10000;
	.@upper_choice = select("不指定 Upper:普通職業（0）:轉生職業（1）:養子職業（2）:取消並返回");
	if (.@upper_choice == 5) {
		goto OnGMJobMenu;
	}

	.@command$ = "@jobchange " + .@job_id;
	if (.@upper_choice > 1)
		.@command$ += " " + (.@upper_choice - 2);

	if (select("執行 GM 指令:取消並返回") == 2) {
		goto OnGMJobMenu;
	}
	getmapxy(.@return_map$, .@return_x, .@return_y, BL_PC);
	close2;
	atcommand .@command$;
	atcommand "@allskill";
	sleep2 300;
	warp .@return_map$, .@return_x, .@return_y;
	end;

S_JobExtraOption:
	.@job_id = getarg(0);

	// Mechanic
	// 推車 / 魔導機甲
	// ========================================================
	if (.@job_id == JOB_MECHANIC) {

		.@choice = select("使用推車:搭乘魔導機甲:都不要");

		switch(.@choice) {
		case 1:
			if (!checkcart())
				setcart;
			break;

		case 2:
			if (
				getskilllv("NC_MADOLICENCE") > 0 &&
				!checkmadogear()
			)
				setmadogear;
			break;
		}

		close;
	}

// ========================================================
	// Meister
	// 推車 / 魔導機甲
	// ========================================================

		.@choice = select("使用推車:搭乘魔導機甲:都不要");

		switch(.@choice) {
		case 1:
			if (!checkcart())
				setcart;
			break;

		case 2:
			if (
				getskilllv("NC_MADOLICENCE") > 0 &&
				!checkmadogear()
			)
				setmadogear;
			break;
		}

		close;
	}

// ========================================================
	// 其他商人系
	// 推車
	// ========================================================
	if (getskilllv("MC_PUSHCART") > 0) {

		if (select("使用推車:不要使用") == 1) {
			if (!checkcart())
				setcart;
		}

		close;
	}

// ========================================================
	// Ranger
	// Falcon / Warg
	// ========================================================
	if (.@job_id == JOB_RANGER) {

		.@choice = select("攜帶獵鷹:取得狼笛:都不要");

		switch(.@choice) {

		case 1:
			if (
				getskilllv("HT_FALCON") > 0 &&
				!checkfalcon() &&
				!checkwug()
			)
				setfalcon;
			break;

		case 2:
